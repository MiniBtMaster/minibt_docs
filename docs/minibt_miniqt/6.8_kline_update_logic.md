class TqApiQTimer(QTimer):
    def __init__(self, main_window):
        super().__init__()
        self.main_window = main_window
        self.update_queue = FixedSizeQueue(10, self.init_update())
        self._last_is_update = self.update_queue.any
        self.timeout.connect(self.wait_update)
        self.start()
    
    def init_update(self):
        """初始化时快速获取数据"""
        count = 0
        start_time = time.time()
        while count < 100 and (time.time() - start_time) < 1.0:
            is_update = self.api.wait_update(1)
            count += 1
            if is_update:
                return is_update
        return False
    
    def wait_update(self):
        """定时器回调：调用 TqApi.wait_update()"""
        is_update = self.main_window.tq_api.wait_update(1)
        if is_update:
            self.update_queue.add(is_update)
        self._last_is_update = self.update_queue.any
```

**关键逻辑：**

- 使用 `QTimer` 定时触发 `wait_update()` 调用
- 初始化时快速获取数据（最多100次或1秒）
- 使用 `FixedSizeQueue` 缓存最近10次更新状态

def get_new_kline(self):
    """处理K线数据"""
    if not self.symbol:
        return self._kline
    
    if self.is_stock:
        # 股票：使用 StockApi

            kline = stock_api.get_quote(self.symbol)
            if kline is not None:

        return self._kline

        kline = self.api.get_kline_serial(self.symbol, self.cycle, len(self.chart.candle_data)).copy()
        # 期货数据需要时区调整（UTC+8）

        kline.columns = FILED.TALL
        return kline

- 期货数据：天勤返回 UTC 时间，需要加上 `8*3.6e12`（8小时的纳秒数）转换为北京时间
- 股票数据：StockApi 返回已转换的时间，无需调整

# stock_api_qtimer.py - StockApiQTimer 类
class StockApiQTimer(QObject):
    """股票API定时器 - 类似天勤的 TqApiQTimer"""
    
    def __init__(self, stock_api: StockApi, parent=None):
        super().__init__(parent)
        self.stock_api = stock_api
        self.timer = QTimer(self)
        self.timer.timeout.connect(self._on_timeout)
        self._running = False
        self._update_interval = 3000  # 默认3秒更新间隔
    
    def start(self):
        """启动定时器"""
        self._running = True
        self._update_interval = self._get_update_interval()
        self.timer.start(self._update_interval)
    
    def _get_update_interval(self):
        """根据交易时段动态调整更新频率"""
        now = datetime.now().time()
        
        # 交易时段（9:30-11:30, 13:00-15:00）：3秒
        if (dt_time(9, 30) <= now <= dt_time(11, 30)) or (dt_time(13, 0) <= now <= dt_time(15, 0)):
            return 3000
        
        # 盘前（9:15-9:30）：10秒
        elif dt_time(9, 15) <= now < dt_time(9, 30):
            return 10000
        
        # 其他时间：30秒
        else:
            return 30000
```

- 交易时段：3秒
- 盘前：10秒
- 其他时间：30秒

#### 步骤 3：获取历史K线并订阅

```python
# stock_api.py - get_kline_serial() (约第146行)
def get_kline_serial(self, symbol: str, period: int, length: int=800) -> pd.DataFrame:
    """
    获取历史K线并注册订阅（使用 MooTDX 数据源）

    Args:
        symbol: 股票代码，支持格式：
                - "000001"（纯代码，自动判断市场）
                - "SH.600000" 或 "SZ.000001"（带交易所前缀）
                - "600000.SH"（天勤格式）
        period: 周期（秒），支持的值：
                - 60: 1分钟
                - 300: 5分钟
                - 900: 15分钟
                - 1800: 30分钟
                - 3600: 1小时
                - 86400: 日线
                - 604800: 周线
                - 2419200: 月线
                - 7776000: 季线
                - 31536000: 年线
        length: K线数量

    Returns:
        DataFrame: columns=['datetime', 'open', 'high', 'low', 'close', 'volume']
                  datetime 为纳秒时间戳（19位）
    """
    # 解析股票代码和市场
    code, market = self._parse_symbol(symbol)

    # === 检查是否已订阅，如果已订阅则直接返回缓存数据 ===
    with self._lock:
        for (sub_code, sub_period), cached_df in self._subscribed_symbols.items():
            if sub_code == code and sub_period == period:
                return cached_df  # 返回缓存数据

    # === 使用 MooTDX 数据源获取数据 ===
    df = pd.DataFrame(columns=['datetime', 'open', 'high', 'low', 'close', 'volume'])

    if self._initialized and self._client:
        # 获取 MooTDX frequency 参数
        frequency = self.PERIOD_TO_FREQUENCY.get(period, 'day')

        try:
            # MooTDX bars() 自动识别市场
            df = self._client.bars(
                symbol=code,
                frequency=frequency,
                offset=length
            )

            if df is not None and not df.empty:
                # 标准化数据格式
                df = self._standardize_kline(df)

        except Exception as e:
            print(f"[StockApi] MooTDX 获取K线失败: {e}")

    # === 注册订阅（如果有数据） ===
    if not df.empty:
        frequency = self.PERIOD_TO_FREQUENCY.get(period, 'day')
        key = (code, period)
        with self._lock:
            self._subscribed_symbols[key] = df.copy()
            self._symbol_configs[key] = {
                "period": period,
                "length": length,
                "frequency": frequency,
                "market": market,
            }

    return df
```

**关键改动：**

- **订阅键改为 tuple**：`(code, period)` 替代原来的 `code`，支持同一股票多周期订阅
- **缓存复用**：已订阅则直接返回缓存数据，避免重复请求
- **PERIOD_TO_FREQUENCY**：使用字符串别名（'1m', '5m', 'day'等）替代整数频率码

**数据标准化：**
```python
# stock_api.py - _standardize_kline() (约第751行)
def _standardize_kline(self, df: pd.DataFrame) -> pd.DataFrame:
    """
    标准化K线数据格式
    
    MooTDX bars() 原始字段：
    - datetime, open, close, high, low, vol, amount, volume, year, month, day, hour, minute
    - vol 为本根K线成交量
    - volume 同 vol
    
    转换为：datetime(纳秒时间戳，19位), open, high, low, close, volume
    """
    # 确保 datetime 是纳秒时间戳（19位）
    if 'datetime' in df.columns:
        if df['datetime'].dtype == 'object':
            df['datetime'] = pd.to_datetime(df['datetime'])
        
        first_val = df['datetime'].iloc[0]
        
        if hasattr(first_val, 'timestamp'):
            # datetime 对象：直接转为纳秒时间戳
            df['datetime'] = df['datetime'].astype('int64')
        elif first_val > 1e18:
            # 已经是纳秒格式（19位），不做转换
            pass
        elif first_val > 1e15:
            # 微秒格式（16位），转为纳秒
            df['datetime'] = df['datetime'] * 1000
        elif first_val > 1e11:
            # 毫秒格式（13位），转为纳秒
            df['datetime'] = df['datetime'] * 1_000_000
        elif first_val < 1e11:
            # 秒格式（10位），转为纳秒
            df['datetime'] = df['datetime'] * 1_000_000_000
    
    # 只保留需要的列（与天勤一致）
    cols_to_keep = ['datetime', 'open', 'high', 'low', 'close', 'volume']
    
    # 使用整数索引（reset_index）
    df = df[cols_to_keep].reset_index(drop=True)
    
    return df
```

**标准化要点：**

- datetime 自动识别多种格式（秒、毫秒、微秒、纳秒）并转换为纳秒时间戳
- 只保留 OHLCV 五字段，与天勤格式完全一致
- 使用整数索引（reset_index）便于后续处理

#### 步骤 4：等待数据更新（简化版 - 全部使用 bars()）

```python
# stock_api.py - wait_update() (约第228行)
def wait_update(self, deadline: float = None) -> bool:
    """
    等待数据更新 - 全部使用 bars() 刷新数据
    
    流程：
    1. 检查初始化状态和交易时段
    2. 保存更新前最后一根K线快照（供 is_changing 对比）
    3. 对每个订阅的 (code, period) 调用 bars() 获取最新数据
    4. 标准化数据并更新缓存
    
    Args:
        deadline: 超时时间（秒），None 表示立即返回
    
    Returns:
        bool: 是否有数据更新
    """
    if not self._initialized or not self._client:
        return False
    
    # 非交易时间直接返回 False
    if not self.is_trading_time():
        return False
    
    with self._lock:
        if not self._subscribed_symbols:
            return False
        
        # 获取所有已订阅的 (code, period) 对
        subscribed_items = list(self._subscribed_symbols.items())
    
    if not subscribed_items:
        return False
    
    # === 保存快照（供 is_changing 比较） ===
    self._last_bar_snapshot.clear()
    for key, df in subscribed_items:
        if not df.empty:
            last = df.iloc[-1]
            self._last_bar_snapshot[key] = {
                'open': float(last['open']),
                'high': float(last['high']),
                'low': float(last['low']),
                'close': float(last['close']),
                'volume': float(last['volume']),
                'datetime': int(last['datetime']),
            }
    
    has_update = False
    
    # === 全部使用 bars() 刷新数据 ===
    for key, _ in subscribed_items:
        code, period = key
        config = self._symbol_configs.get(key, {})
        frequency = config.get('frequency', 'day')
        max_length = config.get('length', 800)
        
        try:
            # 使用 MooTDX bars() 获取最新K线数据
            new_df = self._client.bars(
                symbol=code,
                frequency=frequency,
                offset=max_length
            )
            
            if new_df is not None and not new_df.empty:
                # 标准化数据格式
                new_df = self._standardize_kline(new_df)
                with self._lock:
                    self._subscribed_symbols[key] = new_df
                has_update = True
        except Exception as e:
            print(f"[StockApi.wait_update] bars() 失败 ({code}, {period}s): {e}")
    
    self._has_update = has_update
    return has_update
```

**核心逻辑（简化版）：**

1. **交易时段检测**：非交易时间直接返回 False
2. **保存快照**：更新前保存每个订阅最后一根K线的数据到 `_last_bar_snapshot`
3. **bars() 刷新**：对每个订阅调用 `MooTDX.bars()` 获取最新数据
4. **标准化更新**：使用 `_standardize_kline()` 标准化数据格式后更新缓存

**移除的复杂逻辑：**

- ~~quotes() 批量获取实时行情~~ → 全部使用 bars()
- ~~_is_new_period() 新周期判断~~ → bars() 自动返回完整K线
- ~~_create_new_bar() 新K线生成~~ → bars() 返回标准格式
- ~~_update_current_bar() 当前K线更新~~ → bars() 直接替换
- ~~_changed_symbols, _changed_fields~~ → 使用 _last_bar_snapshot 快照对比

#### 步骤 5：is_changing() 快照对比机制

```python
# stock_api.py - is_changing() (约第299行)
def is_changing(self, obj: Any, key: Union[str, List[str], None] = None) -> bool:
    """
    判断数据是否有更新（仿天勤 TqApi.is_changing）
    
    通过对比 wait_update 前后的 _last_bar_snapshot 实现。
    
    Args:
        obj:
            - str: 股票代码，如 "000001"
            - pd.DataFrame: get_kline_serial 返回的K线数据
        key:
            - None: 检查所有字段 (open, high, low, close, volume)
            - str: 检查指定字段（如 'close'）
            - List[str]: 检查多个字段
    """
    if key is None:
        keys = ['open', 'high', 'low', 'close', 'volume']
    elif isinstance(key, str):
        keys = [key]
    else:
        keys = key
    
    if isinstance(obj, str):
        code = self._parse_symbol(obj)[0]
        return self._is_symbol_changing(code, keys)
    elif isinstance(obj, pd.DataFrame):
        if obj.empty:
            return False
        return self._is_df_changing(obj, keys)
    return False

def _is_df_changing(self, df: pd.DataFrame, keys: List[str]) -> bool:
    """根据 DataFrame 实例精确匹配订阅周期后对比"""
    for key, subscribed_df in self._subscribed_symbols.items():
        if subscribed_df is df:  # 精确匹配 DataFrame 实例
            old_bar = self._last_bar_snapshot.get(key)
            if old_bar is None:
                return True  # 无快照则认为有变化
            last = df.iloc[-1]
            for field in keys:
                if old_bar.get(field, 0) != float(last[field]):
                    return True
            return False
    return False

def _is_symbol_changing(self, code: str, keys: List[str]) -> bool:
    """遍历该股票所有订阅周期，任一有变化即返回 True"""
    for key, df in self._subscribed_symbols.items():
        if key[0] == code and not df.empty:
            old_bar = self._last_bar_snapshot.get(key)
            if old_bar is None:
                return True
            last = df.iloc[-1]
            for field in keys:
                if old_bar.get(field, 0) != float(last[field]):
                    return True
    return False
```

**快照对比流程：**

1. `wait_update()` 在更新数据前，保存每个订阅最后一根K线到 `_last_bar_snapshot`
2. `wait_update()` 使用 `bars()` 刷新数据后，更新 `_subscribed_symbols`
3. `is_changing()` 对比 `_last_bar_snapshot`（旧数据）和 `_subscribed_symbols`（新数据）
4. 任一字段变化则返回 True

### 4.2 StockApiQTimer 的角色定位

`StockApiQTimer` 的作用是**驱动 StockApi.wait_update() 更新数据缓存**，它与 `ChartUpdateManager` 的关系如下：

```
StockApiQTimer (定时调用 StockApi.wait_update())
        │
        ▼ 保存快照到 _last_bar_snapshot
        │
        ▼ bars() 刷新数据 → 更新 _subscribed_symbols 缓存
        │
        ▼
ChartUpdateManager._update_kline() (定时从缓存读取数据)
        │
        ▼ is_changing() 快照对比判断是否有变化
        │
        ▼ 更新图表显示
```

**关键时序（简化版）：**

1. `StockApiQTimer` 每隔 3-30 秒调用 `StockApi.wait_update()`
2. `wait_update()` 保存快照后，使用 `bars()` 刷新 `_subscribed_symbols` 缓存
3. `ChartUpdateManager` 每隔 N 毫秒调用 `chart._fetch_kline_data()`
4. `_fetch_kline_data()` 通过 `stock_api.get_quote()` 读取缓存的K线数据
5. `is_changing()` 对比 `_last_bar_snapshot` 和当前数据判断是否有更新
